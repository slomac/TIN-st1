#!/bin/bash
KOPIE="$HOME/kopie"
mkdir -p "$KOPIE"

LICZBA_PRZED=$(ls -1 "$KOPIE" 2>/dev/null | wc -l)

cp "$HOME"/*.txt "$KOPIE"

LICZBA_PO=$(ls -1 "$KOPIE" 2>/dev/null | wc -l)

ROZNICA=$((LICZBA_PO - LICZBA_PRZED))

echo "Data: $(date)" > "$KOPIE/info.txt"
